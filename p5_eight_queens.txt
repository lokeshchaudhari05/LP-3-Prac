
N = 8

def is_safe(board, row, col):
    for r in range(N):
        c = board[r]
        if c != -1 and (c == col or abs(c - col) == abs(r - row)):
            return False
    return True

def solve(board, row, fixed_row):
    if row == N:
        return True
    if row == fixed_row:                      
        return solve(board, row + 1, fixed_row)
    for col in range(N):
        if is_safe(board, row, col):
            board[row] = col
            if solve(board, row + 1, fixed_row):
                return True
            board[row] = -1                   
    return False

if __name__ == "__main__":
    r0, c0 = 0, 0                             
    board = [-1] * N
    board[r0] = c0
    if solve(board, 0, r0):
        print(f"First queen at (row {r0}, col {c0}). Final 8-Queens matrix:\n")
        for r in range(N):
            print(" ".join("Q" if board[r] == c else "." for c in range(N)))
        print("\nColumn per row:", board)
    else:
        print("No solution with that first queen position")










"""
8-Queens with the first queen fixed
Aim: place 8 queens with no two attacking each other, with the first queen already placed.
Problem: queens attack along rows, columns and diagonals. The first queen is at (0,0).
Method: backtracking. Place one queen per row, try each column, and check safety.
Safety test: the same column, or abs(c - col) == abs(r - row) for a diagonal.
Backtrack: if a row has no safe column, remove the previous queen and move it to the next column.
Complexity: O(N!) in the worst case, reduced by pruning.
Output: columns per row are [0, 4, 7, 5, 2, 6, 1, 3], printed as an 8×8 matrix of Q and dots.
"""
