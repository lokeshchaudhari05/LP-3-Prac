# ---- Cell 1: Import libraries ----
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans

# ---- Cell 2: Load dataset ----
df = pd.read_csv('sales_data_sample.csv', encoding='latin1')
print(df.shape)

# ---- Cell 3: Pre-process and scale ----
data = df[['QUANTITYORDERED', 'PRICEEACH', 'SALES', 'MSRP']].dropna()
X = StandardScaler().fit_transform(data)

# ---- Cell 4: Elbow method ----
wcss = []
for k in range(1, 11):
    km = KMeans(n_clusters=k, random_state=42, n_init=10)
    km.fit(X)
    wcss.append(km.inertia_)

plt.plot(range(1, 11), wcss, marker='o')
plt.xlabel('Number of clusters (K)')
plt.ylabel('WCSS')
plt.title('Elbow Method')
plt.show()

# ---- Cell 5: K-Means clustering (take K from the bend in the elbow graph) ----
kmeans = KMeans(n_clusters=3, random_state=42, n_init=10)
data = data.copy()
data['Cluster'] = kmeans.fit_predict(X)
print(data['Cluster'].value_counts())

# ---- Cell 6: Show the clusters ----
sns.scatterplot(x=data['QUANTITYORDERED'], y=data['SALES'], hue=data['Cluster'], palette='Set1')
plt.title('K-Means Clusters')
plt.show()
