
def knapsack_01(values, weights, W):
    n = len(values)
    dp = [[0] * (W + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        for w in range(W + 1):
            dp[i][w] = dp[i - 1][w]
            if weights[i - 1] <= w:
                dp[i][w] = max(dp[i][w], values[i - 1] + dp[i - 1][w - weights[i - 1]])

    chosen, w = [], W
    for i in range(n, 0, -1):
        if dp[i][w] != dp[i - 1][w]:
            chosen.append(i)
            w -= weights[i - 1]
    return dp[n][W], sorted(chosen)

if __name__ == "__main__":
    values  = [60, 100, 120]
    weights = [10, 20, 30]
    W = 50
    best, items = knapsack_01(values, weights, W)
    print("Values :", values)
    print("Weights:", weights, "| Capacity:", W)
    print("Maximum value:", best)
    print("Items chosen (1-indexed):", items)
